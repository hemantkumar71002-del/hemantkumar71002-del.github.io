Nancy AI

An Android app made for blind and low-vision people. The camera detects objects and speaks their direction, name and distance. It recognises familiar faces, so you know who is in front of you. You control everything by voice, so no screen is needed, and it works offline. It is a helping tool and does not replace a white cane or a guide.

ESP32 with push buttons and an HC-SR04 ultrasonic sensor. It sends button presses and distance to the Rose AI app over Bluetooth (BLE).

The ECHO pin gives 5V but the ESP32 pins take 3.3V. Put a 1k resistor from ECHO to GPIO 18 and a 2k resistor from GPIO 18 to GND. The buttons use the internal pull-up, so they need no resistor.

esp32_rose.ino
#include <Arduino.h>
#include <BLEDevice.h>
#include <BLEServer.h>
#include <BLEUtils.h>
#include <BLE2902.h>

#define DEVICE_NAME "ESP32_Object_Detector"
#define SERVICE_UUID        "4fafc201-1fb5-459e-8fcc-c5c9c331914b"
#define CHARACTERISTIC_UUID "beb5483e-36e1-4688-b7f5-ea07361b26a8"

// Ultrasonic HC-SR04
#define TRIG_PIN 5
#define ECHO_PIN 18

// Push buttons (one side to GPIO, other side to GND)
#define BTN_CAMERA 27
#define BTN_VOICE  13
#define BTN_MIC    14
#define BTN_SOS    25

BLECharacteristic *pCharacteristic = nullptr;
bool deviceConnected = false;
unsigned long lastDistanceTime = 0;
const unsigned long DISTANCE_INTERVAL = 500;

struct Button {
  int pin;
  const char *command;
  int lastReading;
  int stableState;
  unsigned long lastChange;
};

Button buttons[] = {
  {BTN_CAMERA, "BTN:CAM",   HIGH, HIGH, 0},
  {BTN_VOICE,  "BTN:VOICE", HIGH, HIGH, 0},
  {BTN_MIC,    "BTN:MIC",   HIGH, HIGH, 0},
  {BTN_SOS,    "BTN:SOS",   HIGH, HIGH, 0}
};
const int BUTTON_COUNT = sizeof(buttons) / sizeof(buttons[0]);

void sendBLE(String message) {
  if (!deviceConnected || pCharacteristic == nullptr) return;
  pCharacteristic->setValue(message.c_str());
  pCharacteristic->notify();
  Serial.println("BLE sent: " + message);
}

float readDistanceCM() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(3);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  unsigned long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  if (duration == 0) return -1;
  return duration * 0.0343 / 2.0;
}

void checkUltrasonic() {
  if (millis() - lastDistanceTime < DISTANCE_INTERVAL) return;
  lastDistanceTime = millis();

  float distance = readDistanceCM();
  if (distance > 0 && distance <= 400) {
    sendBLE("DIST:" + String((int)round(distance)));
    Serial.print("Distance: ");
    Serial.print(distance);
    Serial.println(" cm");
  } else {
    sendBLE("DIST:NONE");
    Serial.println("Distance: out of range");
  }
}

void checkButtons() {
  for (int i = 0; i < BUTTON_COUNT; i++) {
    int reading = digitalRead(buttons[i].pin);
    if (reading != buttons[i].lastReading) {
      buttons[i].lastChange = millis();
      buttons[i].lastReading = reading;
    }
    if (millis() - buttons[i].lastChange >= 40) {
      if (reading != buttons[i].stableState) {
        buttons[i].stableState = reading;
        if (buttons[i].stableState == LOW) {   // pressed = LOW
          sendBLE(buttons[i].command);
          Serial.println("Button pressed: " + String(buttons[i].command));
        }
      }
    }
  }
}

class RoseServerCallbacks : public BLEServerCallbacks {
  void onConnect(BLEServer *server) override {
    deviceConnected = true;
    Serial.println("Rose AI connected");
  }
  void onDisconnect(BLEServer *server) override {
    deviceConnected = false;
    Serial.println("Rose AI disconnected");
    BLEDevice::startAdvertising();
  }
};

class RoseCharacteristicCallbacks : public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic *characteristic) override {
    String received = String(characteristic->getValue().c_str());
    received.trim();
    Serial.println("Rose AI received: " + received);
  }
};

void setup() {
  Serial.begin(115200);
  delay(500);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  digitalWrite(TRIG_PIN, LOW);

  for (int i = 0; i < BUTTON_COUNT; i++) {
    pinMode(buttons[i].pin, INPUT_PULLUP);
    buttons[i].lastReading = digitalRead(buttons[i].pin);
    buttons[i].stableState = buttons[i].lastReading;
  }

  BLEDevice::init(DEVICE_NAME);
  BLEServer *server = BLEDevice::createServer();
  server->setCallbacks(new RoseServerCallbacks());

  BLEService *service = server->createService(SERVICE_UUID);
  pCharacteristic = service->createCharacteristic(
    CHARACTERISTIC_UUID,
    BLECharacteristic::PROPERTY_READ |
    BLECharacteristic::PROPERTY_WRITE |
    BLECharacteristic::PROPERTY_NOTIFY
  );
  pCharacteristic->addDescriptor(new BLE2902());
  pCharacteristic->setCallbacks(new RoseCharacteristicCallbacks());
  pCharacteristic->setValue("Rose AI Ready");
  service->start();

  BLEAdvertising *advertising = BLEDevice::getAdvertising();
  advertising->addServiceUUID(SERVICE_UUID);
  advertising->setScanResponse(true);
  BLEDevice::startAdvertising();

  Serial.println("ROSE AI ESP32 READY");
}

void loop() {
  checkButtons();
  checkUltrasonic();
  delay(5);
}

Room Drive

An Android app for sharing files inside a company, school or team. The admin creates a room and gives each person a user ID and password. Users sign in with that ID, open PDF, Word, PowerPoint and image files, and upload files if the admin allows it. One user ID works on one device at a time, the admin can remove any ID at any moment, and new files appear live on every connected device.
